-- this depends on the month under consideration
drop TABLE if exists monthly_sums_nov;
CREATE TEMP TABLE monthly_sums_nov AS
SELECT 
    cust_id,
    SUM(CASE WHEN MONTH = 1 THEN TRAN_AMT_IN_AC ELSE 0 END) AS Sum_032025,
    SUM(CASE WHEN MONTH = 2 THEN TRAN_AMT_IN_AC ELSE 0 END) AS Sum_042025,
    SUM(CASE WHEN MONTH = 3 THEN TRAN_AMT_IN_AC ELSE 0 END) AS Sum_052025,
    SUM(CASE WHEN MONTH = 4 THEN TRAN_AMT_IN_AC ELSE 0 END) AS Sum_062025,
    SUM(CASE WHEN MONTH = 5 THEN TRAN_AMT_IN_AC ELSE 0 END) AS Sum_072025,
    SUM(CASE WHEN MONTH = 6 THEN TRAN_AMT_IN_AC ELSE 0 END) AS Sum_082025
FROM b1_dec
GROUP BY cust_id;

drop TABLE if exists BASE_DATA_dec;
drop TABLE if exists BASE_DATA_JAN;
drop TABLE if exists BASE_DATA_feb;
CREATE TABLE BASE_DATA_feb AS
SELECT 
    a.*, 
    CASE 
        WHEN EXTRACT(DAY FROM a.TRAN_DATE) <= 10 OR EXTRACT(DAY FROM a.TRAN_DATE) >= 20 THEN 1 
        ELSE 0 
    END AS DATE_BET_20_10,
    b.Sum_032025,
    b.Sum_042025,
    b.Sum_052025,
    b.Sum_062025,
    b.Sum_072025,
    b.Sum_082025,
    CASE 
        WHEN TO_CHAR(a.TRAN_DATE, 'MMYYYY') = '012026' THEN a.TRAN_AMT_IN_AC / b.Sum_032025
        WHEN TO_CHAR(a.TRAN_DATE, 'MMYYYY') = '022026' THEN a.TRAN_AMT_IN_AC / b.Sum_042025
        WHEN TO_CHAR(a.TRAN_DATE, 'MMYYYY') = '032026' THEN a.TRAN_AMT_IN_AC / b.Sum_052025
        WHEN TO_CHAR(a.TRAN_DATE, 'MMYYYY') = '042026' THEN a.TRAN_AMT_IN_AC / b.Sum_062025
        WHEN TO_CHAR(a.TRAN_DATE, 'MMYYYY') = '052026' THEN a.TRAN_AMT_IN_AC / b.Sum_072025
        WHEN TO_CHAR(a.TRAN_DATE, 'MMYYYY') = '062026' THEN a.TRAN_AMT_IN_AC / b.Sum_082025 
        ELSE 0 
    END AS Px,
    CASE WHEN Px >= 0.4 THEN 1 ELSE 0 END AS Px_GTE_40_PER,
    CASE WHEN a.TRAN_AMT_IN_AC >= 10000 THEN 1 ELSE 0 END AS Tran_GTE_10K
FROM b1_dec a
LEFT JOIN monthly_sums_nov b ON a.cust_id = b.cust_id;

DROP TABLE IF EXISTS merchants_rg_salc;
CREATE TEMP TABLE merchants_rg_salc AS
SELECT *,
    CASE
        WHEN 
            tran_partclr ILIKE '%SALARY%' OR 
            tran_partclr ILIKE '% SALARY %' OR 
            tran_partclr ILIKE '%WAGE%' OR
            tran_partclr ILIKE 'salary%' OR  
            tran_partclr ILIKE 'sal %' OR 
            tran_partclr ILIKE '%/salary%' OR 
            tran_partclr ILIKE '%/salary' OR 
            tran_partclr ILIKE '%/ salary%' OR
            tran_partclr ILIKE '%/ sal %' OR
            tran_partclr ILIKE '%/sal %' OR
            tran_partclr ILIKE '%/ salar%' OR 
            tran_partclr ILIKE '%/salar' OR 
            tran_partclr ILIKE '%-sal' OR  
            tran_partclr ILIKE '%-sal-%' OR     
            tran_partclr ILIKE '%cognizantsal%' OR
            tran_partclr ILIKE '%salar%' OR
            tran_partclr ILIKE '%payroll%' OR
            tran_partclr ILIKE '%WAGES%' OR
            tran_partclr ILIKE '%BULK DEPOSIT%' OR
            tran_partclr ILIKE  '% INCOME %' or
            tran_partclr ILIKE  'PROFESSIONAL FEE%' or
            tran_partclr ILIKE  'STIPEND%' or
            tran_partclr ILIKE  '%PROFESSIONALFEE%' or
            tran_partclr ILIKE  '%PROFESSIONAL FEE%' or
            tran_partclr ILIKE  '%STIPEND%' or 
            tran_partclr ILIKE  '%compensation%'
            THEN 'SALARY'
        WHEN tran_partclr ILIKE '%IFT%' THEN 
            CASE 
                WHEN tran_partclr ILIKE '%IFT-%' THEN SPLIT_PART(tran_partclr, '-', 2)
                ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'IFT\\s*', ''))
            END
        WHEN tran_partclr ILIKE 'FROM %' THEN 
            TRIM(
                SPLIT_PART(tran_partclr, ' ', 2) || 
                COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 3), '') || 
                COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '')
            )
        WHEN tran_partclr ILIKE '%NACH%' THEN 
            CASE 
                 WHEN tran_partclr ILIKE '%NACH-%' THEN SPLIT_PART(tran_partclr, '-', 4)
                 ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'NACH\\s*', ''))
            END
        WHEN tran_partclr ILIKE 'IB:RECEIVED FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE 'FUND TRF FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE 'FT FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE 'FUNDS TRF FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE '%UPI%' THEN 
            CASE 
                 WHEN tran_partclr ILIKE '%UPI/%' THEN SPLIT_PART(tran_partclr, '/', 2)
                 ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'UPI\\s*', ''))
            END
        WHEN tran_partclr ILIKE '%IMPS%' THEN 
            CASE
                WHEN tran_partclr ILIKE '%IMPS/%' THEN SPLIT_PART(tran_partclr, '/', 3)
                ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'IMPS\\s*', ''))
            END
        WHEN tran_partclr ILIKE '%MB:RECEIVED FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE '%IB:FUND%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE '%NEFT%' THEN 
            CASE
                WHEN tran_partclr ILIKE '%NEFT %' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 3) || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 5), '') ||
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), ''))
                ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'NEFT\\s*', ''))
            END
        WHEN tran_partclr ILIKE '%IB: FUND TRANSFER%' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 5) || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 8), ''))
        WHEN tran_partclr ILIKE '%FUND TRANSFER FROM%' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 5) || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 8), ''))
        WHEN tran_partclr ILIKE '%FUNDS TRANSFER FROM%' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 4) || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 5), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), ''))    
        WHEN tran_partclr ILIKE '%FT FROM%' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 3) || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 5), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), ''))
        WHEN tran_partclr ILIKE '%RTGS%' THEN 
            CASE
                WHEN tran_partclr ILIKE '%RTGS %' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 3) || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 5), '') || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), ''))
                ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'RTGS\\s*', ''))
            END
        ELSE TRIM(REGEXP_REPLACE(tran_partclr, '(NEFT|IFT|RTGS|UPI|IMPS|MB):?\\s*', ''))
    END AS remitter
FROM data_3;

-- Cleaning Merchant Names --
DROP TABLE IF exists merchants_rg_sal1c;
Create TEMP table merchants_rg_sal1c as 
SELECT 
    A.*,
    TRIM(REGEXP_REPLACE(LOWER(A.cleaned_string),
        '(neft|imps|rtgs|ift|ft|upi|chq|cash|transfer|trf|mb:|payment|paymen)',
        ' ')) AS Merchant
FROM (
    SELECT *, TRIM(REGEXP_REPLACE(remitter, '[^a-zA-Z\s]', ' ')) AS cleaned_string
    FROM merchants_rg_salc
) AS A;

DROP TABLE IF EXISTS merchants_rg_sal;
CREATE TEMP TABLE merchants_rg_sal AS
SELECT *,
    CASE
        WHEN 
            tran_partclr ILIKE '%SALARY%' OR 
            tran_partclr ILIKE '% SALARY %' OR 
            tran_partclr ILIKE '%WAGE%' OR
            tran_partclr ILIKE 'salary%' OR  
            tran_partclr ILIKE 'sal %' OR 
            tran_partclr ILIKE '%/salary%' OR 
            tran_partclr ILIKE '%/salary' OR 
            tran_partclr ILIKE '%/ salary%' OR
            tran_partclr ILIKE '%/ sal %' OR
            tran_partclr ILIKE '%/sal %' OR
            tran_partclr ILIKE '%/ salar%' OR 
            tran_partclr ILIKE '%/salar' OR 
            tran_partclr ILIKE '%-sal' OR  
            tran_partclr ILIKE '%-sal-%' OR     
            tran_partclr ILIKE '%cognizantsal%' OR
            tran_partclr ILIKE '%salar%' OR
            tran_partclr ILIKE '%payroll%' OR
            tran_partclr ILIKE '%WAGES%' OR
            tran_partclr ILIKE '%BULK DEPOSIT%' OR
            tran_partclr ILIKE  '% INCOME %' or
            tran_partclr ILIKE  'PROFESSIONAL FEE%' or
            tran_partclr ILIKE  'STIPEND%' or
            tran_partclr ILIKE  '%PROFESSIONALFEE%' or
            tran_partclr ILIKE  '%PROFESSIONAL FEE%' or
            tran_partclr ILIKE  '%STIPEND%'
            THEN 'SALARY'
        WHEN tran_partclr ILIKE '%IFT%' THEN 
            CASE 
                WHEN tran_partclr ILIKE '%IFT-%' THEN SPLIT_PART(tran_partclr, '-', 2)
                ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'IFT\\s*', ''))
            END
        WHEN tran_partclr ILIKE 'FROM %' THEN 
            TRIM(
                SPLIT_PART(tran_partclr, ' ', 2) || 
                COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 3), '') || 
                COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '')
            )
        WHEN tran_partclr ILIKE '%NACH%' THEN 
            CASE 
                 WHEN tran_partclr ILIKE '%NACH-%' THEN SPLIT_PART(tran_partclr, '-', 4)
                 ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'NACH\\s*', ''))
            END
        WHEN tran_partclr ILIKE 'IB:RECEIVED FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE 'FUND TRF FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE 'FT FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE 'FUNDS TRF FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE '%UPI%' THEN 
            CASE 
                 WHEN tran_partclr ILIKE '%UPI/%' THEN SPLIT_PART(tran_partclr, '/', 2)
                 ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'UPI\\s*', ''))
            END
        WHEN tran_partclr ILIKE '%IMPS%' THEN 
            CASE
                WHEN tran_partclr ILIKE '%IMPS/%' THEN SPLIT_PART(tran_partclr, '/', 3)
                ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'IMPS\\s*', ''))
            END
        WHEN tran_partclr ILIKE '%MB:RECEIVED FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE '%IB:FUND%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE '%NEFT%' THEN 
            CASE
                WHEN tran_partclr ILIKE '%NEFT %' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 3) || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 5), '') ||
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), ''))
                ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'NEFT\\s*', ''))
            END
        WHEN tran_partclr ILIKE '%IB: FUND TRANSFER%' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 5) || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 8), ''))
        WHEN tran_partclr ILIKE '%FUND TRANSFER FROM%' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 5) || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 8), ''))
        WHEN tran_partclr ILIKE '%FUNDS TRANSFER FROM%' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 4) || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 5), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), ''))    
        WHEN tran_partclr ILIKE '%FT FROM%' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 3) || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 5), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), ''))
        WHEN tran_partclr ILIKE '%RTGS%' THEN 
            CASE
                WHEN tran_partclr ILIKE '%RTGS %' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 3) || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 5), '') || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), ''))
                ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'RTGS\\s*', ''))
            END
        ELSE TRIM(REGEXP_REPLACE(tran_partclr, '(NEFT|IFT|RTGS|UPI|IMPS|MB):?\\s*', ''))
    END AS remitter
FROM data_3M21;

CREATE TEMP TABLE merchants_rg_salp AS
SELECT *,
    CASE
        WHEN 
            tran_partclr ILIKE '%SALARY%' OR 
            tran_partclr ILIKE '% SALARY %' OR 
            tran_partclr ILIKE '%WAGE%' OR
            tran_partclr ILIKE 'salary%' OR  
            tran_partclr ILIKE 'sal %' OR 
            tran_partclr ILIKE '%/salary%' OR 
            tran_partclr ILIKE '%/salary' OR 
            tran_partclr ILIKE '%/ salary%' OR
            tran_partclr ILIKE '%/ sal %' OR
            tran_partclr ILIKE '%/sal %' OR
            tran_partclr ILIKE '%/ salar%' OR 
            tran_partclr ILIKE '%/salar' OR 
            tran_partclr ILIKE '%-sal' OR  
            tran_partclr ILIKE '%-sal-%' OR     
            tran_partclr ILIKE '%cognizantsal%' OR
            tran_partclr ILIKE '%salar%' OR
            tran_partclr ILIKE '%payroll%' OR
            tran_partclr ILIKE '%WAGES%' OR
            tran_partclr ILIKE '%BULK DEPOSIT%' OR
            tran_partclr ILIKE  '% INCOME %' or
            tran_partclr ILIKE  'PROFESSIONAL FEE%' or
            tran_partclr ILIKE  'STIPEND%' or
            tran_partclr ILIKE  '%PROFESSIONALFEE%' or
            tran_partclr ILIKE  '%PROFESSIONAL FEE%' or
            tran_partclr ILIKE  '%STIPEND%'
            THEN 'SALARY'
        WHEN tran_partclr ILIKE '%IFT%' THEN 
            CASE 
                WHEN tran_partclr ILIKE '%IFT-%' THEN SPLIT_PART(tran_partclr, '-', 2)
                ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'IFT\\s*', ''))
            END
        WHEN tran_partclr ILIKE 'FROM %' THEN 
            TRIM(
                SPLIT_PART(tran_partclr, ' ', 2) || 
                COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 3), '') || 
                COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '')
            )
        WHEN tran_partclr ILIKE '%NACH%' THEN 
            CASE 
                 WHEN tran_partclr ILIKE '%NACH-%' THEN SPLIT_PART(tran_partclr, '-', 4)
                 ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'NACH\\s*', ''))
            END
        WHEN tran_partclr ILIKE 'IB:RECEIVED FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE 'FUND TRF FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE 'FT FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE 'FUNDS TRF FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE '%UPI%' THEN 
            CASE 
                 WHEN tran_partclr ILIKE '%UPI/%' THEN SPLIT_PART(tran_partclr, '/', 2)
                 ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'UPI\\s*', ''))
            END
        WHEN tran_partclr ILIKE '%IMPS%' THEN 
            CASE
                WHEN tran_partclr ILIKE '%IMPS/%' THEN SPLIT_PART(tran_partclr, '/', 3)
                ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'IMPS\\s*', ''))
            END
        WHEN tran_partclr ILIKE '%MB:RECEIVED FROM%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE '%IB:FUND%' THEN TRIM(SPLIT_PART(tran_partclr, 'FROM', 2))
        WHEN tran_partclr ILIKE '%NEFT%' THEN 
            CASE
                WHEN tran_partclr ILIKE '%NEFT %' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 3) || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 5), '') ||
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), ''))
                ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'NEFT\\s*', ''))
            END
        WHEN tran_partclr ILIKE '%IB: FUND TRANSFER%' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 5) || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 8), ''))
        WHEN tran_partclr ILIKE '%FUND TRANSFER FROM%' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 5) || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 8), ''))
        WHEN tran_partclr ILIKE '%FUNDS TRANSFER FROM%' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 4) || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 5), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), ''))    
        WHEN tran_partclr ILIKE '%FT FROM%' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 3) || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 5), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
            COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), ''))
        WHEN tran_partclr ILIKE '%RTGS%' THEN 
            CASE
                WHEN tran_partclr ILIKE '%RTGS %' THEN TRIM(SPLIT_PART(tran_partclr, ' ', 3) || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 4), '') || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 5), '') || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 6), '') || 
                    COALESCE(' ' || SPLIT_PART(tran_partclr, ' ', 7), ''))
                ELSE TRIM(REGEXP_REPLACE(tran_partclr, 'RTGS\\s*', ''))
            END
        ELSE TRIM(REGEXP_REPLACE(tran_partclr, '(NEFT|IFT|RTGS|UPI|IMPS|MB):?\\s*', ''))
    END AS remitter
FROM Data4_m22;

DROP TABLE IF EXISTS valid_txns_step1p;
CREATE TEMP TABLE valid_txns_step1p AS
WITH base AS (
    SELECT *
    FROM sandbox.percent_test1_rg
    WHERE cluster_id IS NOT NULL
),
ranked_txns AS (
    SELECT *,
           ROW_NUMBER() OVER (
               PARTITION BY cust_id, cluster_id, month
               ORDER BY tran_amt_in_ac DESC, tran_date ASC
           ) AS rn
    FROM base
),
max_per_month AS (
    SELECT *
    FROM ranked_txns
    WHERE rn = 1
),
valid_customers AS (
    SELECT cust_id, cluster_id
    FROM max_per_month
    GROUP BY cust_id, cluster_id
    HAVING COUNT(DISTINCT month) >= 3
       AND COUNT(*) BETWEEN 3 AND 7
)
SELECT m.*
FROM max_per_month m
JOIN valid_customers v
  ON m.cust_id = v.cust_id
 AND m.cluster_id = v.cluster_id;

DROP TABLE IF EXISTS flagged_txns_step2p;
CREATE TEMP TABLE flagged_txns_step2p AS
WITH date_flags AS (
    SELECT vt.*,
        EXTRACT(DAY FROM tran_date) AS trans_day,
        EXTRACT(DAY FROM FIRST_VALUE(tran_date) OVER (
            PARTITION BY cust_id 
            ORDER BY tran_date 
            ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
        )) AS base_day,
        CASE 
            WHEN LEAST(
                ABS(EXTRACT(DAY FROM tran_date) - EXTRACT(DAY FROM FIRST_VALUE(tran_date) OVER (
                    PARTITION BY cust_id 
                    ORDER BY tran_date 
                    ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
                ))),
                31 - ABS(EXTRACT(DAY FROM tran_date) - EXTRACT(DAY FROM FIRST_VALUE(tran_date) OVER (
                    PARTITION BY cust_id 
                    ORDER BY tran_date 
                    ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
                )))
            ) <= 5
            THEN 1 ELSE 0
        END AS day_diff_in_5
    FROM valid_txns_step1p vt
),
median_amt_per_cluster AS (
    SELECT cust_id, cluster_id, PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY tran_amt_in_ac) AS median_amt
    FROM date_flags
    GROUP BY cust_id, cluster_id
),
amount_flags AS (
    SELECT df.*, m.median_amt,
           CASE WHEN ABS(tran_amt_in_ac - m.median_amt) <= 0.2 * m.median_amt THEN 1 ELSE 0 END AS range_status,
           CASE WHEN ABS(tran_amt_in_ac - m.median_amt) <= 0.5 * m.median_amt THEN 1 ELSE 0 END AS range_status_50_per
    FROM date_flags df
    JOIN median_amt_per_cluster m
      ON df.cust_id = m.cust_id
     AND df.cluster_id = m.cluster_id
)
SELECT * FROM amount_flags;

-- ============================================================================
-- 1) CREATE FINAL STRINGS TABLE (one method per cust_id, then take ALL rows)
--    Table: sandbox.salary_strings_final1_dec25
-- ============================================================================
DROP TABLE IF EXISTS sandbox.salary_strings_final1_july26;
CREATE TABLE sandbox.salary_strings_final1_july26 AS
WITH
-- Detect which methods have data for each cust_id
salary_custs AS (SELECT DISTINCT cust_id FROM sandbox.salary_test2_rg),
percent_custs AS (SELECT DISTINCT cust_id FROM sandbox.percent_test2_rg),
company_custs AS (SELECT DISTINCT cust_id FROM sandbox.company_test2_rg),
method_candidates AS (
    SELECT cust_id, 'Salary'  AS method_name, 1 AS priority FROM salary_custs UNION ALL
    SELECT cust_id, 'Percent' AS method_name, 2 AS priority FROM percent_custs UNION ALL
    SELECT cust_id, 'Company' AS method_name, 3 AS priority FROM company_custs),
chosen_method AS (
    SELECT cust_id, MIN(priority) AS chosen_priority
    FROM method_candidates
    GROUP BY cust_id),
-- Bring all rows from the chosen method table for each cust_id
salary_rows AS (
    SELECT s.*, 'Salary' AS chosen_method, 1 AS method_priority,
           'sandbox.salary_test2_rg' AS source_table
    FROM sandbox.salary_test2_rg s
    JOIN chosen_method cm ON cm.cust_id = s.cust_id AND cm.chosen_priority = 1),
percent_rows AS (
    SELECT p.*, 'Percent' AS chosen_method, 2 AS method_priority,
           'sandbox.percent_test2_rg' AS source_table
    FROM sandbox.percent_test2_rg p
    JOIN chosen_method cm ON cm.cust_id = p.cust_id AND cm.chosen_priority = 2),
company_rows AS (
    SELECT c.*, 'Company' AS chosen_method, 3 AS method_priority,
           'sandbox.company_test2_rg' AS source_table
    FROM sandbox.company_test2_rg c
    JOIN chosen_method cm ON cm.cust_id = c.cust_id AND cm.chosen_priority = 3),
latest_month AS (
    SELECT cust_id, MAX(month) AS max_month
    FROM (
        SELECT cust_id, month FROM sandbox.salary_test2_rg UNION ALL
        SELECT cust_id, month FROM sandbox.percent_test2_rg UNION ALL
        SELECT cust_id, month FROM sandbox.company_test2_rg) all_months
    GROUP BY cust_id)

SELECT f.*, CASE WHEN f.month = 6 THEN 1 ELSE 0 END AS latest_month_flag
FROM (
    SELECT * FROM salary_rows UNION ALL
    SELECT * FROM percent_rows UNION ALL
    SELECT * FROM company_rows) f
JOIN latest_month lm ON f.cust_id = lm.cust_id;
